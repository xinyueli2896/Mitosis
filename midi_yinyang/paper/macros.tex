% ---------------------------------------------------------------
% System names, defined once. NEVER type a system name literally in
% the body or in a table -- change it here and it changes everywhere,
% including every table cell and figure caption.
%
% Put \input{macros} in the preamble of the main file, after
% \usepackage{xspace}.
%
% xspace is what makes "\sys is" render as "Mitosis is" rather than
% "Mitosisis": LaTeX swallows the space after a control word, and
% xspace puts it back -- except before punctuation, where you do not
% want one. It leaves the space before "(", so \sys(ours) gives
% "Mitosis (ours)".
% ---------------------------------------------------------------

% ---- ours ------------------------------------------------------
\newcommand{\sys}{Mitosis\xspace}
% No trailing space: for table cells, captions, and anywhere the next
% character is set by the table rather than by the sentence.
\newcommand{\sysbare}{Mitosis}
\newcommand{\sysours}{\sysbare~(ours)}

% ---- baselines -------------------------------------------------
% Spell a baseline the way its own paper does, and only here.
